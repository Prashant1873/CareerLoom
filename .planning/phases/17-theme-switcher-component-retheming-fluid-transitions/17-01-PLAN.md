<task id="17-01-01">
  <read_first>
    - frontend/index.html
  </read_first>
  <action>
    Add blocking inline theme script to `frontend/index.html` head:
    ```html
    <script>
      (function() {
        try {
          var saved = localStorage.getItem('careerloom_theme');
          var theme = saved || (window.matchMedia('(prefers-color-scheme: light)').matches ? 'light' : 'dark');
          document.documentElement.setAttribute('data-theme', theme);
        } catch (e) {}
      })();
    </script>
    ```
  </action>
  <verification>
    Inspect `frontend/index.html` to confirm inline theme initialization script is placed prior to `#root`.
  </verification>
</task>
